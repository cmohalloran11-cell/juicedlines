# Pinned to the exact versions this app is developed/tested against (2026-08 production-
# readiness audit finding: loose >= pins with no lockfile meant a fresh `pip install` could
# silently pull a newer, breaking major/minor version of anything here at any time -- real
# risk given fastapi and psycopg both ship fast). `~=X.Y.Z` allows patch-level security/bug
# fixes to install automatically (e.g. 0.138.1 -> 0.138.2) while blocking any minor/major
# bump, which is where a breaking API change is actually likely to land. Bump deliberately,
# not accidentally: change a pin here, run the full test suite, then commit.
fastapi~=0.141.1
uvicorn[standard]~=0.52.1
requests~=2.34.2
urllib3~=2.7.0
numpy~=2.5.1
# scipy powers the vendored stat-projector engine's Monte-Carlo (projector/models/montecarlo.py).
# Without it the board silently falls back to the empirical median for every MLB prop.
scipy~=1.18.0
# Server-side Supabase JWT verification (auth.py). Fail-closed if unset.
pyjwt~=2.13.0
# Postgres backend for the store/ layer (DATABASE_URL → Postgres; SQLite otherwise).
psycopg[binary]~=3.3.4
# Error monitoring (main.py). Fully optional -- only activates if SENTRY_DSN is set; a
# local dev run with no env vars never talks to Sentry at all. Installed unconditionally
# (small, pure-Python, no native deps) so it's always available if an operator sets the DSN.
sentry-sdk[fastapi]~=2.66.1
